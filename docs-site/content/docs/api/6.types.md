# Types & Interfaces

```ts
import type {
  OidcClaimMapping,
  OidcConfig,
  OidcFeatures,
  OidcProvider,
  OidcUserContext,
} from "ioserver-oidc";
```

## `OidcProvider`

```ts
type OidcProvider = "auth-service" | "oidc";
```

## `OidcConfig`

```ts
interface OidcConfig {
  provider?: OidcProvider;
  authServiceUrl: string;
  appSlug: string;
  jwksUri?: string;
  issuer?: string;
  audience?: string;
  algorithms?: readonly string[];
  clockToleranceSeconds?: number;
  allowMachineTokens?: boolean;
  adminRoles?: readonly string[];
  gatewayUrl?: string;
  claims?: OidcClaimMapping;
}
```

## `OidcUserContext`

```ts
interface OidcUserContext {
  userId: string;
  sub: string;
  email: string | null;
  name: string | null;
  userRole: string;
  roles: string[];
  permissions: string[];
  features: OidcFeatures;
  org_id?: string;
  plan?: string;
  sid?: string;
  clientId?: string;
  scopes?: string[];
  isMachine?: boolean;
  emailVerified?: boolean;
  picture?: string | null;
  company?: string | null;
}
```

## `OidcFeatures`

```ts
type OidcFeatures = Record<string, unknown>;
```

## `OidcClaimMapping`

```ts
interface OidcClaimMapping {
  roles: string;
  permissions: string;
  features: string;
  plan: string;
  orgId: string;
  email: string;
  name: string;
  picture: string;
  company: string;
  authorizedParty: readonly string[];
}
```

## Augmenting Fastify and Socket.IO types

```ts
// src/types/ioserver-oidc.d.ts
import "fastify";
import "socket.io";
import type { OidcFeatures, OidcUserContext } from "ioserver-oidc";

declare module "fastify" {
  interface FastifyRequest {
    sub: string;
    userId: string;
    userRole: string;
    roles: string[];
    permissions: string[];
    features: OidcFeatures;
    plan: string | null;
    auth: OidcUserContext;
    authToken: string;
  }
}

declare module "socket.io" {
  interface Socket {
    sub: string;
    userId: string;
    userRole: string;
    roles: string[];
    permissions: string[];
    features: OidcFeatures;
    plan: string | null;
    auth: OidcUserContext;
    authToken: string;
  }
}
```
