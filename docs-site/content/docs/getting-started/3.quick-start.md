# Quick Start

```ts
import { IOServer, BaseController, BaseService } from "ioserver";
import {
  OidcConfigManager,
  OidcHttpMiddleware,
  OidcSocketMiddleware,
  OidcSocketAdminMiddleware,
} from "ioserver-oidc";

class ProfileController extends BaseController {
  async getMe(request: any, reply: any) {
    reply.send({
      userId: request.userId,
      role: request.userRole,
      plan: request.plan,
      permissions: request.permissions,
    });
  }
}

class ChatService extends BaseService {
  async message(socket: any, data: { text: string }, callback?: Function) {
    socket.broadcast.emit("chat:message", {
      from: socket.userId,
      text: data.text,
    });
    if (callback) callback({ sent: true });
  }
}

class AdminService extends BaseService {
  async listUsers(socket: any, _data: unknown, callback?: Function) {
    if (callback) callback({ users: [] });
  }
}

const server = new IOServer({ host: "0.0.0.0", port: 3000 });

server.addManager({ name: "oidcConfig", manager: OidcConfigManager });

server.addController({
  name: "profile",
  controller: ProfileController,
  middlewares: [OidcHttpMiddleware],
});

server.addService({
  name: "chat",
  service: ChatService,
  middlewares: [OidcSocketMiddleware],
});

server.addService({
  name: "admin",
  service: AdminService,
  middlewares: [OidcSocketMiddleware, OidcSocketAdminMiddleware],
});

await server.start();
```

Route file `routes/profile.json`:

```json
[{ "method": "GET", "url": "/me", "handler": "getMe" }]
```

## Test the endpoint

```bash
curl -H "Authorization: Bearer $TOKEN" http://localhost:3000/profile/me
# → { "userId": "…", "role": "user", "plan": "pro", "permissions": [] }

curl http://localhost:3000/profile/me
# → { "statusCode": 401, "code": "ERR_AUTH_TOKEN_REQUIRED", … }
```

## Test the WebSocket connection

```ts
import { io } from "socket.io-client";

const socket = io("http://localhost:3000/chat", {
  auth: { token: accessToken },
});

socket.on("connect_error", (err) => console.error(err.message));
```

## Registration order

Register `OidcConfigManager` before the controllers and services. IOServer calls
`manager.start()` during startup, and the middlewares resolve their configuration
from `appHandle.oidcConfig` on first use. Without the manager they fall back to
`process.env`, which works but skips the startup validation.
